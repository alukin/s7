include(${PROJECT_SOURCE_DIR}/s7hardware/cmake/OptionalBackend.cmake)
s7_optional_backend(SOAPYSDR "SoapySDR>=0.8" TRUE)
add_library(s7sdr SHARED src/sdr.cpp)
target_link_libraries(s7sdr PUBLIC s7hardware)
target_include_directories(s7sdr PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(S7_HAVE_SOAPYSDR)
    target_sources(s7sdr PRIVATE src/soapy.cpp)
    target_link_libraries(s7sdr PRIVATE PkgConfig::S7_DEP_SOAPYSDR)
    target_compile_definitions(s7sdr PRIVATE S7_HAVE_SOAPYSDR=1)
endif()
set_target_properties(s7sdr PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(UNIX AND NOT APPLE)
    set_target_properties(s7sdr PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
s7_enable_project_warnings(s7sdr)
add_library(s7sdr_plugin MODULE src/plugin.cpp)
target_link_libraries(s7sdr_plugin PRIVATE s7sdr)
set_target_properties(s7sdr_plugin PROPERTIES OUTPUT_NAME s7sdr-elements INSTALL_RPATH "$ORIGIN/../..")
install(TARGETS s7sdr_plugin LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/s7/plugins COMPONENT s7hardware)
install(TARGETS s7sdr EXPORT s7hardwareTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT ${S7_HARDWARE_INSTALL_COMPONENT}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${S7_HARDWARE_INSTALL_COMPONENT} NAMELINK_COMPONENT s7sdk
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT s7sdk)
install(DIRECTORY include/s7 DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} COMPONENT s7sdk)
s7_add_test_executable(s7sdr_smoke tests/sdr_smoke.cpp)
target_link_libraries(s7sdr_smoke PRIVATE s7sdr)
s7_enable_project_warnings(s7sdr_smoke)
add_test(NAME s7sdr_smoke COMMAND s7sdr_smoke)
if(S7_HAVE_SOAPYSDR)
    s7_add_test_executable(s7sdr_soapy_smoke tests/soapy_smoke.cpp)
    target_link_libraries(s7sdr_soapy_smoke PRIVATE s7sdr s7io s7radio PkgConfig::S7_DEP_SOAPYSDR)
    target_compile_definitions(s7sdr_soapy_smoke PRIVATE S7_FM_EXAMPLE="${PROJECT_SOURCE_DIR}/examples/sdr/fm_receiver.s7diagram")
    s7_enable_project_warnings(s7sdr_soapy_smoke)
    add_test(NAME s7sdr_soapy_smoke COMMAND s7sdr_soapy_smoke)
endif()
