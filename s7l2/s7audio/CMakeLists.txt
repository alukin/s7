include(${PROJECT_SOURCE_DIR}/s7hardware/cmake/OptionalBackend.cmake)
s7_optional_backend(PORTAUDIO portaudio-2.0 TRUE)
set(pipewire_supported FALSE)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(pipewire_supported TRUE)
endif()
s7_optional_backend(PIPEWIRE "libpipewire-0.3>=0.3.65" ${pipewire_supported})
add_library(s7audio SHARED src/audio.cpp)
target_link_libraries(s7audio PUBLIC s7hardware)
target_include_directories(s7audio PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
foreach(backend PORTAUDIO PIPEWIRE)
    if(S7_HAVE_${backend})
        string(TOLOWER ${backend} source)
        target_sources(s7audio PRIVATE src/${source}.cpp)
        target_link_libraries(s7audio PRIVATE PkgConfig::S7_DEP_${backend})
        target_compile_definitions(s7audio PRIVATE S7_HAVE_${backend}=1)
    endif()
endforeach()
set_target_properties(s7audio PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
if(UNIX AND NOT APPLE)
    set_target_properties(s7audio PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
s7_enable_project_warnings(s7audio)
add_library(s7audio_plugin MODULE src/plugin.cpp)
target_link_libraries(s7audio_plugin PRIVATE s7audio)
set_target_properties(s7audio_plugin PROPERTIES OUTPUT_NAME s7audio-elements INSTALL_RPATH "$ORIGIN/../..")
install(TARGETS s7audio_plugin LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/s7/plugins COMPONENT s7hardware)
install(TARGETS s7audio EXPORT s7hardwareTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT ${S7_HARDWARE_INSTALL_COMPONENT}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${S7_HARDWARE_INSTALL_COMPONENT} NAMELINK_COMPONENT s7sdk
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT s7sdk)
install(DIRECTORY include/s7 DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} COMPONENT s7sdk)
s7_add_test_executable(s7audio_smoke tests/audio_smoke.cpp)
target_link_libraries(s7audio_smoke PRIVATE s7audio)
target_include_directories(s7audio_smoke PRIVATE src)
s7_enable_project_warnings(s7audio_smoke)
add_test(NAME s7audio_smoke COMMAND s7audio_smoke)
if(S7_HAVE_PIPEWIRE)
    s7_add_test_executable(s7audio_pipewire_smoke tests/pipewire_smoke.cpp src/pipewire.cpp)
    target_link_libraries(s7audio_pipewire_smoke PRIVATE s7hardware)
    target_include_directories(s7audio_pipewire_smoke PRIVATE include)
    target_include_directories(s7audio_pipewire_smoke SYSTEM PRIVATE
        $<TARGET_PROPERTY:PkgConfig::S7_DEP_PIPEWIRE,INTERFACE_INCLUDE_DIRECTORIES>)
    s7_enable_project_warnings(s7audio_pipewire_smoke)
    add_test(NAME s7audio_pipewire_smoke COMMAND s7audio_pipewire_smoke)
endif()
if(S7_HAVE_PORTAUDIO)
    s7_add_test_executable(s7audio_portaudio_smoke tests/portaudio_smoke.cpp src/portaudio.cpp)
    target_link_libraries(s7audio_portaudio_smoke PRIVATE s7hardware)
    target_include_directories(s7audio_portaudio_smoke PRIVATE include)
    target_include_directories(s7audio_portaudio_smoke SYSTEM PRIVATE
        $<TARGET_PROPERTY:PkgConfig::S7_DEP_PORTAUDIO,INTERFACE_INCLUDE_DIRECTORIES>)
    s7_enable_project_warnings(s7audio_portaudio_smoke)
    add_test(NAME s7audio_portaudio_smoke COMMAND s7audio_portaudio_smoke)
endif()
