cmake_minimum_required(VERSION 3.20)
project(s7wavelet VERSION 1.0.0 LANGUAGES CXX)
find_package(s7core 0.1 REQUIRED)
find_package(s7spectral CONFIG REQUIRED)
add_library(s7wavelet MODULE src/plugin_wavelet.cpp)
target_include_directories(s7wavelet PRIVATE include ${S7SPECTRAL_INCLUDE_DIR})
target_link_libraries(s7wavelet PRIVATE s7::s7core s7spectral::analysis)
s7_embed_metadata(s7wavelet resources/element_metadata.json wavelet_metadata)
target_compile_options(s7wavelet PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS s7wavelet LIBRARY DESTINATION lib/s7/plugins)
install(DIRECTORY include/ DESTINATION include COMPONENT s7sdk)

include(CTest)
if(BUILD_TESTING)
    add_executable(wavelet_analysis_smoke tests/analysis_smoke.cpp)
    target_include_directories(wavelet_analysis_smoke PRIVATE include ${S7SPECTRAL_INCLUDE_DIR})
    target_link_libraries(wavelet_analysis_smoke PRIVATE s7::s7core s7spectral::analysis)
    add_test(NAME wavelet_analysis_smoke COMMAND wavelet_analysis_smoke)
endif()

add_library(s7wavelet_analysis INTERFACE)
add_library(s7wavelet::analysis ALIAS s7wavelet_analysis)
set_target_properties(s7wavelet_analysis PROPERTIES EXPORT_NAME analysis)
target_include_directories(s7wavelet_analysis INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(s7wavelet_analysis INTERFACE s7::s7core s7spectral::analysis)
install(TARGETS s7wavelet_analysis EXPORT s7waveletTargets COMPONENT s7sdk)
install(EXPORT s7waveletTargets NAMESPACE s7wavelet:: DESTINATION lib/cmake/s7wavelet COMPONENT s7sdk)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/s7waveletConfig.cmake [=[
include(CMakeFindDependencyMacro)
find_dependency(s7core 0.1)
find_dependency(s7spectral CONFIG)
include("${CMAKE_CURRENT_LIST_DIR}/s7waveletTargets.cmake")
]=])
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/s7waveletConfig.cmake DESTINATION lib/cmake/s7wavelet COMPONENT s7sdk)
